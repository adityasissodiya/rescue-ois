% M2: incident-journal sequencing. The command edge accepts each
% client_event_id once, assigns event_seq atomically, and folds duplicates
% onto the existing entry; consumers replay in event_seq order.
\begin{tikzpicture}[
  every node/.style={font=\scriptsize},
  box/.style={draw, rounded corners=1pt, align=center, inner sep=2.5pt, minimum height=7mm},
  inbox/.style={box, fill=gray!8, text width=19mm},
  proc/.style={box, fill=gray!15, text width=22mm},
  cell/.style={draw, minimum width=6.5mm, minimum height=6mm, inner sep=1pt},
  result/.style={box, text width=46mm},
  flow/.style={->, >=Stealth, semithick},
  dup/.style={->, >=Stealth, semithick, dashed}
]
\node[inbox] (ops) at (0,0)   {Authority-bearing ops\\(\textit{client\_event\_id})};
\node[proc] (cmd) at (2.55,0) {Command edge\\accept once \textperiodcentered{} assign \textit{event\_seq}};
% totally ordered journal
\node[cell] (c1) at (4.75,0) {1};
\node[cell] (c2) at (5.4,0)  {2};
\node[cell] (c3) at (6.05,0) {3};
\node[cell] (c4) at (6.7,0)  {4};
\node[font=\tiny] at (5.72,0.55) {incident journal (totally ordered)};
\node[result] (mat) at (3.6,-1.7) {Consumers replay in \textit{event\_seq} order $\rightarrow$ materialized state (rebuildable)};

\draw[flow] (ops) -- (cmd);
\draw[flow] (cmd) -- (c1.west);
\draw[flow] (c4) -- (c4|-mat.north) -- (mat.east);
\draw[dup] (cmd.south) .. controls (3.1,-0.85) and (4.8,-0.85) .. (c2.south)
  node[pos=0.5, below, font=\tiny, align=center, text width=30mm]
  {duplicate \textit{client\_event\_id} maps to existing entry (no new seq)};
\end{tikzpicture}
